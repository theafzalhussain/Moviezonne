# ── Everyone else ──────────────────────────────────────────────────────────
# One group for User-agent: * instead of two. Two separate "*" groups is legal
# but fragile — crawlers differ on whether they merge them or honour only the
# first, and the rules below are the ones that actually matter.
#
# No Crawl-delay. Google ignores it, but Bing and Yandex obey it, and at one
# request per second a 1,200-page catalogue takes 20+ minutes of continuous
# crawling to get through. There is no reason to throttle them.
User-agent: *
Content-Signal: search=yes,ai-train=no,use=reference
Allow: /
Disallow: /api/
Disallow: /sw.js
# Search result URLs are an unbounded duplicate space. The homepage also
# serves these with a noindex tag; this keeps them out of the crawl entirely.
Disallow: /*?search=
Disallow: /*&search=
